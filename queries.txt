SELECT SUM(total_amount) AS total_order_amount

PART 3 — JOINS
--------------

21. Display customers and their orders
SELECT 
    c.customer_name,
    o.order_id,
    o.order_date,
    o.total_amount
FROM customers c
JOIN orders o
ON c.customer_id = o.customer_id;

22. Display order details with customer names and restaurant names
SELECT 
    o.order_id,
    c.customer_name,
    r.restaurant_name,
    o.total_amount,
    o.order_status
FROM orders o
JOIN customers c
ON o.customer_id = c.customer_id
JOIN restaurants r
ON o.restaurant_id = r.restaurant_id;

PART 4 — GROUP BY & HAVING
---------------------------

31. Find total orders placed by each customer
SELECT 
    c.customer_name,
    COUNT(o.order_id) AS total_orders
FROM customers c
JOIN orders o
ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name;

32. Find total sales generated by each restaurant
SELECT 
    r.restaurant_name,
    SUM(o.total_amount) AS total_sales
FROM restaurants r
JOIN orders o
ON r.restaurant_id = o.restaurant_id
GROUP BY r.restaurant_id, r.restaurant_name;

33. Find average rating of each restaurant from reviews
SELECT 
    r.restaurant_name,
    AVG(rv.rating) AS average_rating
FROM restaurants r
JOIN reviews rv
ON r.restaurant_id = rv.restaurant_id
GROUP BY r.restaurant_id, r.restaurant_name;

34. Display restaurants having average review rating greater than 4
SELECT 
    r.restaurant_name,
    AVG(rv.rating) AS average_rating
FROM restaurants r
JOIN reviews rv
ON r.restaurant_id = rv.restaurant_id
GROUP BY r.restaurant_id, r.restaurant_name
HAVING AVG(rv.rating) > 4;

35. Find total quantity of food ordered for each food item
SELECT 
    f.food_name,
    SUM(oi.quantity) AS total_quantity
FROM food_items f
JOIN order_items oi
ON f.food_id = oi.food_id
GROUP BY f.food_id, f.food_name;

PART 5 — SUBQUERIES
-------------------

36. Find food items whose price is greater than the average food price
SELECT food_name, price
FROM food_items
WHERE price > (
    SELECT AVG(price)
    FROM food_items
);

37. Find the restaurant having the highest rating
SELECT restaurant_name, rating
FROM restaurants
WHERE rating = (
    SELECT MAX(rating)
    FROM restaurants
);

38. Find customers who placed orders above the average order amount
SELECT DISTINCT customer_name
FROM customers
WHERE customer_id IN (
    SELECT customer_id
    FROM orders
    WHERE total_amount > (
        SELECT AVG(total_amount)
        FROM orders
    )
);

PART 6 — ADVANCED / PROJECT-LEVEL
---------------------------------

39. Display order details with discount applied
SELECT 
    o.order_id,
    c.customer_name,
    o.total_amount,
    oc.discount_amount,
    o.total_amount - oc.discount_amount AS final_amount
FROM orders o
JOIN customers c
ON o.customer_id = c.customer_id
JOIN order_coupons oc
ON o.order_id = oc.order_id;

40. Find the most expensive food item in each restaurant
SELECT 
    r.restaurant_name,
    f.food_name,
    f.price
FROM restaurants r
JOIN food_items f
ON r.restaurant_id = f.restaurant_id
WHERE f.price = (
    SELECT MAX(f2.price)
    FROM food_items f2
    WHERE f2.restaurant_id = f.restaurant_id
);

47. TRIGGER – Prevent Negative Food Price
DELIMITER //

CREATE TRIGGER check_food_price
BEFORE INSERT ON food_items
FOR EACH ROW
BEGIN
    IF NEW.price < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Food price cannot be negative';
    END IF;
END //

DELIMITER ;

Test:

INSERT INTO food_items
VALUES (16,1,1,'Test Food',-100,'Available');

It will show an error because price cannot be negative.

48. TRIGGER – Prevent Invalid Rating
DELIMITER //

CREATE TRIGGER check_review_rating
BEFORE INSERT ON reviews
FOR EACH ROW
BEGIN
    IF NEW.rating < 1 OR NEW.rating > 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 5';
    END IF;
END //

DELIMITER ;

Test:

INSERT INTO reviews
VALUES
(16,1,1,7,'Test review','2026-09-20');
